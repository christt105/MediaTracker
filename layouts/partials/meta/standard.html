{{/* Overrides hugo-blog-awesome's meta/standard.html. The only change versus upstream is the image block: covers are page bundle resources, so the original ".Params.image | absURL" produced a URL relative to the site root that did not exist. Here we resolve the resource and emit its real absolute permalink so social cards show the cover. */}} {{ .Title }} | {{ .Site.Title }} {{/* Description: front matter description, then overview, then site default */}} {{- $description := "" -}} {{- if .Params.description -}} {{- $description = trim .Params.description "\n" -}} {{- else if .Params.overview -}} {{- $description = trim .Params.overview "\n" | truncate 200 -}} {{- else if .Site.Params.description -}} {{- $description = trim .Site.Params.description "\n" -}} {{- end -}} {{ range .AllTranslations }} {{ end }} {{/* og:type and og:url (replaces Hugo's internal opengraph.html, not called) */}} {{- if and .IsPage (not .Date.IsZero) }} {{- end }} {{/* Image: prefer the page bundle resource (resized, real permalink). Fall back to a string image param, then the site ogimage param. */}} {{- $imageURL := "" -}} {{- $imageW := 0 -}} {{- $imageH := 0 -}} {{- with .Params.image -}} {{- with $.Resources.GetMatch . -}} {{- $r := .Resize "600x webp" -}} {{- $imageURL = $r.Permalink -}} {{- $imageW = $r.Width -}} {{- $imageH = $r.Height -}} {{- else -}} {{- $imageURL = . | absURL -}} {{- end -}} {{- else -}} {{- with $.Site.Params.ogimage -}} {{- $imageURL = . | absURL -}} {{- end -}} {{- end -}} {{- if $imageURL -}} {{- if $imageW }} {{- end }} {{- else -}} {{- end -}}
